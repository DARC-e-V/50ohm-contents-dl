        \begin{axis}[
            axis lines=center,
            axis line style={-Triangle},
            scale only axis,

            width=\linewidth*\getDarcImageFactor,
            xmin=0,

            ticks=none,
            xlabel={$t$},
            xlabel style={anchor=north},

        % Input signal: f = 1.1 kHz
        % 2*pi on the x-axis corresponds to 1 ms
        \addplot[
            smooth,

            thick,
            domain=0:24*pi,
            samples=1800
        ]
        {0.8*sin(deg(1.1*x))};

        \draw[dashed, thick]
            (axis cs:8.5*pi,-1.0)
            -- (axis cs:8.5*pi,1.15);

        \draw[dashed, thick]
            (axis cs:10.5*pi,-1.0)
            -- (axis cs:10.5*pi,1.15);

        % Gate-time arrow
        \draw[Triangle-Triangle, thick]
            (axis cs:8.5*pi,1.18)
            -- node[above] {kurze Torzeit $T_\mathrm{G}=\qty{1}{\milli\second}$}
            (axis cs:10.5*pi,1.18);

        % Counted positive zero crossing
        \foreach \k in {5}{
            \addplot[
                only marks,
                mark=*,
                mark size=1.8pt,
                DARCblue
            ]
            coordinates {({20/11*\k*pi},0)};
        }

        \node[anchor=north]
            at (axis cs:9.5*pi,-1.0)
            {$N=1 \quad\Rightarrow\quad f_\mathrm{M}=\dfrac{N}{T_\mathrm{G}}=\qty{1}{\kilo\hertz}$};

        \end{axis}
    \end{scope}
